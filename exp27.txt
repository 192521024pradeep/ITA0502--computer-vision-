import cv2
import os
import urllib.request

# Download Haar Cascade if missing
xml_file = "haarcascade_frontalface_default.xml"

if not os.path.exists(xml_file):
    url = "https://raw.githubusercontent.com/opencv/opencv/4.x/data/haarcascades/haarcascade_frontalface_default.xml"
    urllib.request.urlretrieve(url, xml_file)

# Read image
img = cv2.imread("input.jpg")

if img is None:
    print("Image not found!")
    exit()

# Convert to grayscale
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Load face detector
face = cv2.CascadeClassifier(xml_file)

if face.empty():
    print("Error: XML file could not be loaded!")
    exit()

# Detect faces
faces = face.detectMultiScale(gray, 1.1, 5)

print("Faces detected:", len(faces))

# Draw rectangles
for (x, y, w, h) in faces:
    cv2.rectangle(img, (x, y), (x+w, y+h), (0, 255, 0), 2)

# Display result
cv2.imshow("Face Detection", img)

cv2.waitKey(0)
cv2.destroyAllWindows()
