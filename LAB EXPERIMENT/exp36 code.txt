
import cv2
import numpy as np

# Read input image
img = cv2.imread("input.jpg")

if img is None:
    print("Error: Image not found!")
    exit()

# Convert BGR to HSV
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)

# Define background color range (green)
lower = np.array([35, 40, 40])
upper = np.array([85, 255, 255])

# Create background mask
mask = cv2.inRange(hsv, lower, upper)

# Invert mask to remove background
result = cv2.bitwise_and(img, img, mask=cv2.bitwise_not(mask))

# Display results
cv2.imshow("Original Image", img)
cv2.imshow("Background Mask", mask)
cv2.imshow("Background Removed", result)

# Save result
cv2.imwrite("background_removed.jpg", result)

print("Background subtraction completed!")

cv2.waitKey(0)
cv2.destroyAllWindows()
