
import cv2
import os

# Set image folder
folder = r"C:\COMPUTER VISION\exp29"

# Read input image
img = cv2.imread(os.path.join(folder, "input.jpg"))

if img is None:
    print("Error: Image not found!")
    exit()

# Load Haar Cascade files
face_cascade = cv2.CascadeClassifier(
    os.path.join(folder, "haarcascade_frontalface_default.xml")
)

eye_cascade = cv2.CascadeClassifier(
    os.path.join(folder, "haarcascade_eye.xml")
)

if face_cascade.empty() or eye_cascade.empty():
    print("Error: Cascade XML file not loaded!")
    exit()

# Convert image into grayscale
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Detect faces
faces = face_cascade.detectMultiScale(
    gray,
    scaleFactor=1.1,
    minNeighbors=4,
    minSize=(30, 30)
)

print("Number of faces detected:", len(faces))

# Detect eyes
total_eyes = 0

for (x, y, w, h) in faces:

    # Draw face rectangle
    cv2.rectangle(img, (x, y), (x+w, y+h), (255, 0, 0), 2)

    # Select upper face region
    roi_gray = gray[y:y+int(h*0.65), x:x+w]
    roi_color = img[y:y+int(h*0.65), x:x+w]

    # Resize region
    resized = cv2.resize(roi_gray, None, fx=2, fy=2)

    # Detect eyes
    eyes = eye_cascade.detectMultiScale(
        resized,
        scaleFactor=1.05,
        minNeighbors=3,
        minSize=(10, 10)
    )

    print("Eyes detected:", len(eyes))

    total_eyes += len(eyes)

    # Draw eye rectangles
    for (ex, ey, ew, eh) in eyes:

        ex = ex // 2
        ey = ey // 2
        ew = ew // 2
        eh = eh // 2

        cv2.rectangle(
            roi_color,
            (ex, ey),
            (ex+ew, ey+eh),
            (0, 255, 0),
            2
        )

print("Total eyes detected:", total_eyes)

# Display output
cv2.imshow("Eye Detection Using OpenCV", img)

cv2.waitKey(0)
cv2.destroyAllWindows()
