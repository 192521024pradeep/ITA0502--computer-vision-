
import cv2
import numpy as np

# Get image size from user
height = int(input("Enter image height: "))
width = int(input("Enter image width: "))

# Create a white image
img = np.ones((height, width, 3), dtype=np.uint8) * 255

# Calculate box size (1/10th of image size)
box_h = height // 10
box_w = width // 10

# Create four colored boxes at each corner

# Top-left corner - Black
img[0:box_h, 0:box_w] = [0, 0, 0]

# Top-right corner - Blue
img[0:box_h, width-box_w:width] = [255, 0, 0]

# Bottom-left corner - Green
img[height-box_h:height, 0:box_w] = [0, 255, 0]

# Bottom-right corner - Red
img[height-box_h:height, width-box_w:width] = [0, 0, 255]

# Display image
cv2.imshow("White Image with Four Colored Boxes", img)

# Save image
cv2.imwrite("colored_boxes.jpg", img)

print("Image created successfully!")

cv2.waitKey(0)
cv2.destroyAllWindows()
